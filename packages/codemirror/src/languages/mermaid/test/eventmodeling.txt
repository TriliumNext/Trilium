entity Shop.Cart
tf 02 cmd Shop.AddItem ->> 01 [[CartData]] `json` { "a": { "b": 1 } }
rf 03 evt ItemAdded "inline" %% a comment
timeframe 1 readmodel A ->> 1 ->> 1
resetframe 2 processor B 'x'
