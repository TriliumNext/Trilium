# Groups, postfixes, exceptions and special sequences

railroad-ebnf-beta
  title Numbers
  number = [ "-" ], digit, { digit } ;
  digit ::= "0" | '1' | ? any digit ? ;
  letter = (a | b)+ - "x" c* d? ;
