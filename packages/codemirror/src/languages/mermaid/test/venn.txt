VennDiagram(DiagramName,Title(Keyword,TitleText),
  Set(Keyword,Name),Text(Keyword,Name),Text(Keyword,Name),
  Set(Keyword,Name),Text(Keyword,Name),
  Union(Keyword,Name,",",Name,Label))

# Labels, sizes and quoted names

venn-beta
set A["Alpha"]: 20
set "Quoted set"[Beta]
union A,"Quoted set": .5
  text "note"[Note]
text A,B shared["Shared"] %% a comment

==>

VennDiagram(DiagramName,
  Set(Keyword,Name,Label,Size(":",Number)),
  Set(Keyword,String,Label),
  Union(Keyword,Name,",",String,Size(":",Number)),
  Text(Keyword,String,Label),
  Text(Keyword,Name,",",Name,Name,Label),LineComment)

  Style(Keyword,Name,",",Name,StyleField(Property,":",Number),",",StyleField(Property,":",String),",",StyleField(Property,":",Color)))
